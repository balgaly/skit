# skit CLI Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `skit`, a cross-platform npm CLI package manager for AI agent skills that installs, links, updates, and shares skills from any source.

**Architecture:** CommonJS Node.js CLI. Core modules (`src/core/`) handle config, manifest, linking, scanning, git, and importing. Agent adapters (`src/agents/`) abstract target directories. Commands (`src/commands/`) wire core logic to the CLI. UI helpers (`src/ui/`) provide interactive prompts and formatting. Entry point (`bin/skit.js`) uses commander to route commands.

**Tech Stack:** Node.js (CommonJS), commander (CLI), inquirer (prompts), chalk (colors), ora (spinners), node:test (testing)

```javascript
const fs = require('fs');
const path = require('path');
const os = require('os');

const DEFAULT_CONFIG = { agent: 'claude-code', user: '', skitHome: null };

function getSkitHome() {
  return process.env.SKIT_HOME || path.join(os.homedir(), '.skit');
}

function getConfigPath() {
  return path.join(getSkitHome(), 'config.json');
}

function loadConfig() {
  const configPath = getConfigPath();
  if (!fs.existsSync(configPath)) {
    return { ...DEFAULT_CONFIG };
  }
  return JSON.parse(fs.readFileSync(configPath, 'utf-8'));
}

function saveConfig(config) {
  const configPath = getConfigPath();
  fs.mkdirSync(path.dirname(configPath), { recursive: true });
  fs.writeFileSync(configPath, JSON.stringify(config, null, 2) + '\n');
}

function getConfigValue(key) {
  const config = loadConfig();
  return config[key];
}

function setConfigValue(key, value) {
  const config = loadConfig();
  config[key] = value;
  saveConfig(config);
}

function initConfig() {
  const configPath = getConfigPath();
  if (!fs.existsSync(configPath)) {
    saveConfig({ ...DEFAULT_CONFIG });
  }
}

module.exports = {
  getSkitHome, getConfigPath, loadConfig, saveConfig,
  getConfigValue, setConfigValue, initConfig,
};
```

```javascript
const fs = require('fs');
const path = require('path');
const { getSkitHome } = require('./config');

function getManifestPath() {
  return path.join(getSkitHome(), 'manifest.json');
}

function emptyManifest() {
  return { version: 1, sources: {}, skills: {} };
}

function loadManifest() {
  const p = getManifestPath();
  if (!fs.existsSync(p)) return emptyManifest();
  return JSON.parse(fs.readFileSync(p, 'utf-8'));
}

function saveManifest(manifest) {
  const p = getManifestPath();
  fs.mkdirSync(path.dirname(p), { recursive: true });
  fs.writeFileSync(p, JSON.stringify(manifest, null, 2) + '\n');
}

function addSource(name, data) {
  const m = loadManifest();
  m.sources[name] = data;
  saveManifest(m);
  return m;
}

function removeSource(name) {
  const m = loadManifest();
  delete m.sources[name];
  for (const [sk, sv] of Object.entries(m.skills)) {
    if (sv.source === name) delete m.skills[sk];
  }
  saveManifest(m);
  return m;
}

function addSkill(name, data) {
  const m = loadManifest();
  m.skills[name] = data;
  saveManifest(m);
  return m;
}

function removeSkill(name) {
  const m = loadManifest();
  delete m.skills[name];
  saveManifest(m);
  return m;
}

function getSkill(name) {
  return loadManifest().skills[name] || null;
}

function getSource(name) {
  return loadManifest().sources[name] || null;
}

function listSkills() {
  return loadManifest().skills;
}

function listSources() {
  return loadManifest().sources;
}

function getSkillsBySource(sourceName) {
  const skills = loadManifest().skills;
  const result = {};
  for (const [name, skill] of Object.entries(skills)) {
    if (skill.source === sourceName) result[name] = skill;
  }
  return result;
}

module.exports = {
  getManifestPath, emptyManifest, loadManifest, saveManifest,
  addSource, removeSource, addSkill, removeSkill,
  getSkill, getSource, listSkills, listSources, getSkillsBySource,
};
```

```javascript
const fs = require('fs');
const path = require('path');

function linkSkill(sourcePath, targetPath) {
  const resolvedSource = path.resolve(sourcePath);
  const resolvedTarget = path.resolve(targetPath);
  fs.mkdirSync(path.dirname(resolvedTarget), { recursive: true });

  try {
    const stats = fs.lstatSync(resolvedTarget);
    if (stats.isSymbolicLink()) {
      fs.unlinkSync(resolvedTarget);  // Safe for symlinks + NTFS junctions
    } else if (stats.isDirectory()) {
      throw new Error(`Target is a real directory, not a symlink: ${resolvedTarget}. Remove it manually.`);
    }
  } catch (e) {
    if (e.code !== 'ENOENT') throw e;
  }

  if (process.platform === 'win32') {
    fs.symlinkSync(resolvedSource, resolvedTarget, 'junction');
  } else {
    fs.symlinkSync(resolvedSource, resolvedTarget, 'dir');
  }
}

function unlinkSkill(targetPath) {
  const resolved = path.resolve(targetPath);
  try {
    const stats = fs.lstatSync(resolved);
    if (stats.isSymbolicLink()) {
      fs.unlinkSync(resolved);  // Works for both symlinks and NTFS junctions
    } else if (stats.isDirectory()) {
      throw new Error(`Target is a real directory, not a symlink: ${resolved}`);
    }
  } catch (e) {
    if (e.code !== 'ENOENT') throw e;
  }
}

function isLinked(targetPath) {
  try {
    return fs.lstatSync(path.resolve(targetPath)).isSymbolicLink();
  } catch {
    return false;
  }
}

function getLinkTarget(targetPath) {
  try {
    return fs.readlinkSync(path.resolve(targetPath));
  } catch {
    return null;
  }
}

module.exports = { linkSkill, unlinkSkill, isLinked, getLinkTarget };
```

```javascript
const fs = require('fs');
const path = require('path');

function scanForSkills(dir, adapter) {
  const results = [];
  if (!fs.existsSync(dir)) return results;
  const entries = fs.readdirSync(dir, { withFileTypes: true });
  for (const entry of entries) {
    if (!entry.isDirectory()) continue;
    const fullPath = path.join(dir, entry.name);
    if (adapter.detectSkill(fullPath)) {
      const meta = adapter.getSkillMeta(fullPath);
      results.push({
        name: meta.name || entry.name,
        description: meta.description || '',
        path: entry.name,
        fullPath,
      });
    }
  }
  return results;
}

function parseFrontmatter(content) {
  const match = content.match(/^---\r?\n([\s\S]*?)\r?\n---/);
  if (!match) return {};
  const lines = match[1].split(/\r?\n/);
  const result = {};
  for (const line of lines) {
    const colonIdx = line.indexOf(':');
    if (colonIdx > 0) {
      const key = line.slice(0, colonIdx).trim();
      const value = line.slice(colonIdx + 1).trim();
      result[key] = value;
    }
  }
  return result;
}
```

```javascript
const chalk = require('chalk');

function formatSkillList(skills, sources) {
  if (Object.keys(skills).length === 0) return chalk.dim('  No skills installed.');
  const grouped = {};
  for (const [name, skill] of Object.entries(skills)) {
    if (!grouped[skill.source]) grouped[skill.source] = [];
    grouped[skill.source].push({ name, ...skill });
  }
  const lines = [];
  for (const [src, list] of Object.entries(grouped)) {
    const info = sources[src];
    lines.push('');
    lines.push(chalk.bold(`  ${src}`) + chalk.dim(` (${info?.type || '?'}, ${list.length} skill${list.length === 1 ? '' : 's'})`));
    for (const s of list) lines.push(`    ${chalk.green(s.name)}`);
  }
  return lines.join('\n');
}

function formatDoctorReport(report) {
  const lines = [];
  if (report.brokenLinks.length) {
    lines.push('', chalk.red.bold('  Broken links:'));
    for (const i of report.brokenLinks) lines.push(`    ${chalk.red(i.skill)} -> ${chalk.dim(i.reason)}`);
  }
  if (report.updatesAvailable.length) {
    lines.push('', chalk.yellow.bold('  Updates available:'));
    for (const i of report.updatesAvailable) lines.push(`    ${chalk.yellow(i.source)}: ${i.detail}`);
  }
  if (report.unusedSources.length) {
    lines.push('', chalk.dim.bold('  Unused sources:'));
    for (const i of report.unusedSources) lines.push(`    ${chalk.dim(i)}: cloned but no skills installed`);
  }
  if (!lines.length) lines.push('', chalk.green('  Everything looks good!'));
  return lines.join('\n');
}
```

```javascript
const path = require('path');
const fs = require('fs');
const chalk = require('chalk');
const { getSkitHome } = require('../core/config');
const manifest = require('../core/manifest');
const linker = require('../core/linker');
const git = require('../core/git');
const { detectUrlType, downloadGist, downloadRawFile } = require('../core/importer');
const { getAgentAdapter } = require('../index');
const { createSpinner } = require('../ui/spinner');
const { confirm } = require('../ui/picker');

async function importCommand(url) {
  const adapter = getAgentAdapter();
  const home = getSkitHome();
  const targetDir = adapter.skillDir();
  const detected = detectUrlType(url);

  console.log(chalk.dim(`  Detected: ${detected.type}`));

  if (detected.type === 'github-repo') {
    return require('./install')(`https://github.com/${detected.user}/${detected.repo}`, {});
  }

  const standaloneDir = path.join(home, 'sources', 'external', '_standalone');
  let skillDir, skillName;

  if (detected.type === 'github-gist') {
    const spinner = createSpinner('Downloading gist...');
    spinner.start();
    try {
      const tmpName = `gist-${detected.id.slice(0, 8)}`;
      const tmpDir = path.join(standaloneDir, tmpName);
      const result = await downloadGist(detected.id, tmpDir);
      skillName = result.description
        ? result.description.split(/\s/)[0].toLowerCase().replace(/[^a-z0-9-]/g, '')
        : tmpName;
      if (!skillName) skillName = tmpName;
      skillDir = path.join(standaloneDir, skillName);
      if (tmpDir !== skillDir) {
        if (fs.existsSync(skillDir)) fs.rmSync(skillDir, { recursive: true, force: true });
        fs.renameSync(tmpDir, skillDir);
      }
      spinner.succeed(`Downloaded gist (${result.files.length} files)`);
    } catch (err) { spinner.fail(`Failed: ${err.message}`); return; }

  } else if (detected.type === 'github-subfolder') {
    const repoUrl = `https://github.com/${detected.user}/${detected.repo}`;
    const repoDir = path.join(home, 'sources', 'external', detected.repo);
    if (!fs.existsSync(repoDir)) {
      const spinner = createSpinner(`Cloning ${detected.repo}...`);
      spinner.start();
      try { git.gitClone(repoUrl, repoDir); spinner.succeed(`Cloned ${detected.repo}`); }
      catch (err) { spinner.fail(`Clone failed: ${err.message}`); return; }
    }
    skillDir = path.join(repoDir, detected.path);
    skillName = path.basename(detected.path);
    if (!fs.existsSync(skillDir)) { console.error(chalk.red(`Path not found: ${detected.path}`)); return; }

    if (!manifest.getSource(detected.repo)) {
      manifest.addSource(detected.repo, {
        type: 'external', origin: repoUrl,
        localPath: path.join('sources', 'external', detected.repo),
        gitSha: git.isGitRepo(repoDir) ? git.getGitSha(repoDir) : null,
        addedAt: new Date().toISOString(), updatedAt: new Date().toISOString(),
      });
    }
    linker.linkSkill(skillDir, path.join(targetDir, skillName));
    manifest.addSkill(skillName, {
      source: detected.repo, path: detected.path, agent: adapter.name,
      installedAt: new Date().toISOString(), importedFrom: url,
    });
    console.log(chalk.green(`\n  Installed ${skillName} from ${detected.repo}/${detected.path}`));
    return;

  } else {
    const spinner = createSpinner('Downloading...');
    spinner.start();
    try {
      const urlObj = new URL(url);
      skillName = path.basename(urlObj.pathname, path.extname(urlObj.pathname)) || 'imported-skill';
      skillDir = path.join(standaloneDir, skillName);
      await downloadRawFile(url, skillDir);
      spinner.succeed('Downloaded.');
    } catch (err) { spinner.fail(`Failed: ${err.message}`); return; }
  }

  if (!adapter.detectSkill(skillDir)) {
    const wrap = await confirm('No SKILL.md found. Create a wrapper?', true);
    if (wrap) {
      fs.writeFileSync(path.join(skillDir, 'SKILL.md'),
        `---\nname: ${skillName}\ndescription: Imported from ${url}\n---\n\n# ${skillName}\n`);
    }
  }

  if (!manifest.getSource('_standalone')) {
    manifest.addSource('_standalone', {
      type: 'external', origin: null,
      localPath: path.join('sources', 'external', '_standalone'),
      addedAt: new Date().toISOString(), updatedAt: new Date().toISOString(),
    });
  }

  linker.linkSkill(skillDir, path.join(targetDir, skillName));
  manifest.addSkill(skillName, {
    source: '_standalone', path: skillName, agent: adapter.name,
    installedAt: new Date().toISOString(), importedFrom: url,
  });
  console.log(chalk.green(`\n  Installed ${skillName}`));
}

module.exports = importCommand;
```

```javascript
async function importProfile(file) {
  let profile;
  if (file.startsWith('http://') || file.startsWith('https://')) {
    profile = JSON.parse(await fetchUrl(file));
  } else {
    profile = JSON.parse(fs.readFileSync(path.resolve(file), 'utf-8'));
  }
  console.log(chalk.bold(`\n  Importing profile from ${profile.user || 'unknown'}...`));
  const installCmd = require('./install');

    if (!source.origin) continue;
    if (manifest.getSource(source.name)) {
      console.log(chalk.dim(`  Source '${source.name}' exists, skipping.`));
      continue;
    }
    await installCmd(source.origin, { own: source.type === 'own' });
  }
}

async function diffProfile(target) {
  let profile;
  if (fs.existsSync(path.resolve(target))) {
    profile = JSON.parse(fs.readFileSync(path.resolve(target), 'utf-8'));
  } else {
    profile = await fetchProfileForUser(target);
  }

  const mySkills = manifest.listSkills();
  const theirSkills = {};
  for (const s of profile.skills) theirSkills[s.name] = s;

  const missing = [], onlyMine = [];
  for (const [n, s] of Object.entries(theirSkills)) {
    if (!mySkills[n]) missing.push({ name: n, source: s.source });
  }
  for (const n of Object.keys(mySkills)) {
    if (!theirSkills[n]) onlyMine.push({ name: n, source: mySkills[n].source });
  }

  console.log(chalk.bold(`\n  Profile diff vs ${profile.user || target}:\n`));
  if (missing.length) {
    console.log(chalk.green.bold(`  Missing (${missing.length}):`));
    for (const s of missing) console.log(`    ${chalk.green('+')} ${s.name} ${chalk.dim(`@${s.source}`)}`);
  }
  if (onlyMine.length) {
    console.log(chalk.red.bold(`\n  Only you (${onlyMine.length}):`));
    for (const s of onlyMine) console.log(`    ${chalk.red('-')} ${s.name} ${chalk.dim(`@${s.source}`)}`);
  }
  if (!missing.length && !onlyMine.length) console.log(chalk.green('  Profiles match!'));
}

async function pushProfile() {
  try { execFileSync('gh', ['--version'], { stdio: 'pipe' }); }
  catch {
    console.error(chalk.red('GitHub CLI (gh) not found. Install: https://cli.github.com'));
    console.log(chalk.dim("Run 'skit profile export > skit-profile.json' and create a gist manually."));
    process.exit(1);
  }

  const profile = buildProfile();
  const json = JSON.stringify(profile, null, 2);
  const desc = `skit profile for ${profile.user || 'anonymous'} - ${profile.skills.length} skills`;

  const tmp = path.join(require('os').tmpdir(), 'skit-profile-push.json');
  fs.writeFileSync(tmp, json);
  try {
    const result = execFileSync('gh', ['gist', 'create', '--filename', 'skit-profile.json', '--desc', desc, tmp],
      { encoding: 'utf-8', stdio: ['pipe', 'pipe', 'pipe'] });
    console.log(chalk.green('\n  Profile pushed!'));
    console.log(chalk.dim(`  ${result.trim()}`));
    if (profile.user) console.log(chalk.bold(`\n  Share: npx skit clone ${profile.user}`));
  } catch (err) {
    console.error(chalk.red(`Push failed: ${err.message}`));
  } finally {
    fs.rmSync(tmp, { force: true });
  }
}

async function fetchProfileForUser(username) {
  const data = JSON.parse(await fetchUrl(`https://api.github.com/users/${username}/gists`));
  for (const gist of data) {
    if (gist.files['skit-profile.json']) {
      return JSON.parse(await fetchUrl(gist.files['skit-profile.json'].raw_url));
    }
  }
  throw new Error(`No skit profile found for '${username}'.`);
}
```

```javascript
const chalk = require('chalk');
const manifest = require('../core/manifest');
const { fetchProfileForUser } = require('./profile');
const { fetchUrl } = require('../core/importer');
const installCmd = require('./install');
const { confirm } = require('../ui/picker');

async function cloneCommand(target) {
  let profile;

  if (target.startsWith('http://') || target.startsWith('https://')) {
    try { profile = JSON.parse(await fetchUrl(target)); }
    catch (err) { console.error(chalk.red(`Failed to fetch: ${err.message}`)); process.exit(1); }
  } else {
    try {
      console.log(chalk.dim(`  Fetching ${target}'s profile...`));
      profile = await fetchProfileForUser(target);
    } catch {
      console.error(chalk.red(`No skit profile found for '${target}'.`));
      console.log(chalk.dim('  Users share profiles with: skit profile push'));
      process.exit(1);
    }
  }

  console.log(chalk.bold(`\n  Found ${profile.skills.length} skills from ${profile.sources.length} sources:\n`));
  for (const source of profile.sources) {
    const skills = profile.skills.filter(s => s.source === source.name);
    console.log(chalk.bold(`  ${source.name}`) + chalk.dim(` (${skills.length} skills)`));
    console.log(`    ${skills.map(s => s.name).join(', ')}`);
  }

  const standalone = profile.skills.filter(s => s.source === '_standalone');
  if (standalone.length) {
    console.log(chalk.bold('  standalone') + chalk.dim(` (${standalone.length} skills)`));
    console.log(`    ${standalone.map(s => s.name).join(', ')}`);
  }

  console.log('');
  if (!(await confirm(`Install all ${profile.skills.length} skills?`, true))) return;

  for (const source of profile.sources) {
    if (!source.origin) continue;
    if (manifest.getSource(source.name)) console.log(chalk.dim(`  '${source.name}' already cloned.`));
    await installCmd(source.origin, { own: source.type === 'own' });
  }

  for (const skill of standalone) {
    if (skill.importedFrom) {
      await require('./import')(skill.importedFrom);
    }
  }

  console.log(chalk.green(`\n  Cloned ${profile.user || target}'s profile!`));
}

module.exports = cloneCommand;
```
